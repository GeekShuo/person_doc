代码题，计算 a 的 b/c次方 不能直接用 power 函数 
思路 拆分为 a 的 b 次方，然后开 c 次方幂，a的 b 次方就是用快速幂，开 c 次方，就是枚举 x,比较x的 c 次方跟 a^b的大小，然后二分
```python
from collections import defaultdict, deque, Counter
from typing import List, Dict, Set, Tuple, Optional

def quick_power(base: float, exp: int) -> float:
    """快速幂，计算 base^exp，exp是非负整数"""
    result = 1.0
    b = base
    e = exp
    while e > 0:
        if e % 2 == 1:
            result *= b
        b *= b
        e = e // 2
    return result

def m_root(val: float, m: int, eps=1e-6) -> float:
    """二分法求 val 的 m 次方根 val^(1/m)"""
    if val == 0:
        return 0.0
    # 负数情况：m必须是奇数
    neg = False
    if val < 0:
        if m % 2 == 0:
            raise ValueError("负数不能开偶次方根")
        neg = True
        val = -val
    # 确定二分区间
    low, high = 0.0, max(1.0, val)
    while high - low > eps:
        mid = (low + high) / 2
        mid_pow = quick_power(mid, m)
        if mid_pow < val:
            low = mid
        else:
            high = mid
    res = (low + high) / 2
    return -res if neg else res

def solution():
    # 示例输入，你可以改成读取输入
    x = 4
    n = 1
    m = 2
    # x^(n/m) = (x^n)^(1/m)
    xn = quick_power(x, n)
    ans = m_root(xn, m)
    # 保留2位小数
    ans = round(ans, 2)
    print(ans)

if __name__ == "__main__":
    solution()
```
