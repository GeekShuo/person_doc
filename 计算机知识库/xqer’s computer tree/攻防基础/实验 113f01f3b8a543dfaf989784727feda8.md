```
{
  char * dev;
  char *net_c; //字符串形式网络地址，用于打印输出

  struct tm *now_tm;
  time_t now;

  bpf_u_int32 mask = 0;

  dev = NULL;
  memset(errbuf,0,PCAP_ERRBUF_SIZE);

  //pcap_lookupdev 返回设备名称
  dev = pcap_lookupdev(errbuf);
  if (dev == NULL) {
    fprintf(stderr, "Couldn't find default device: %s\n", errbuf);
    return(2);
  }
  printf("Device: %s\n", dev);
  //lookupnet获得指定网络设备的网络号和掩码
  if (pcap_lookupnet(dev, &net, &mask, errbuf) == -1) {
    fprintf(stderr, "Can't get netmask for device %s\n", dev);
    net_c = 0;
    mask_c = 0;
    return(2);
   }
   //转换网络地址
  addr.s_addr = net;
  net_c = inet_n1toa(addr);
  printf("Net: %s\n", net_c);

  addr.s_addr = mask;
  mask_c = inet_ntoa(addr);
  printf("Mask: %s\n",mask_c);

  printf("==================================================\n");
  //pcap_open_live打开设备文件准备读取数据
  handle = pcap_open_live(dev, BUFSIZ, 1, 1000, errbuf);
  if (handle == NULL) {
    fprintf(stderr, "Couldn't open device %s: %s\n", dev, errbuf);
    return(2);
  }
  //编译过滤规则
  if (pcap_compile(handle, &fp, "ip", 1, mask) == -1) {
    fprintf(stderr, "Couldn't parse filter %s: %s\n", filter_exp, pcap_geterr(handle));
    return(2);
  }
  //设置过滤规则
  if (pcap_setfilter(handle, &fp) == -1) {
     fprintf(stderr, "Couldn't install filter %s: %s\n", filter_exp, pcap_geterr(handle));
     return(2);
   }
   //捕获数据包
   //pcap_loop循环抓取网络数据报文 采用回调函数实现
  if(pcap_loop(handle,-1,callback,NULL)<0)
  {
    (void)fprintf(stderr,"pcap_loop:%s\n",pcap_geterr(handle));
    exit(0);
  }
  //关闭库
  pcap_close(handle);
  return(0);
}
```

# 简易流量分析系统

对给定的协议流量报文能够解析出相关协议类型及相关字段， 比如：http URL、Header等等

```
from scapy.all import *
import re
headers = {}
def extract_http_headers(pcap_file):
    # Open pcap file and read packets
    packets = rdpcap(pcap_file)

    # Regular expression for HTTP headers
    http_header_re = re.compile(b'\r\n\r\n')

    # Dictionary to store headers
    # Iterate through packets
    for packet in packets:
       # Check if packet has a TCP layer
        #print(packet.payload.payload.dport)
        if TCP in packet:
            #print(type(packet))
            # Check if packet has an HTTP layer
            if packet.payload.payload.dport==80: #.haslayer():
                # Extract raw payload from packet
                #print(packet.payload.payload.show())
                payload = bytes(packet[TCP].payload)

                if len(payload)==0:
                    continue
                #print(payload)
                # Split payload into headers and body
                header_data = http_header_re.split(payload, maxsplit=1)[0]

                # # Decode header data and split into lines
                header_lines = header_data.decode().split("\r\n")
                for line in header_lines:
                    #if("Host" in line ):
                    print(line)

    return header_lines
headers = extract_http_headers('1.cap')
```

packet是一个网络层报文，包含了IP，TCP，HTTP信息，packet.payload.payload是TCP报文，过滤下目的地址是否为80端口

之后找到连续的两个回车换行进行分割，前面那部分就是header部分，再把header按行分割
