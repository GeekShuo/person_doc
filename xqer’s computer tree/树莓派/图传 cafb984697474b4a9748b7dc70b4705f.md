import urllib.request
import json
import base64
import datetime
def getByte(path):
    with open(path, 'rb') as f:
        img_byte = base64.b64encode(f.read()) #二进制读取后变base64编码
    img_str = img_byte.decode('ascii') #转成python的unicode
    return img_str

url='http://api.heclouds.com/devices/1062975501/datapoints'
APIKEY ='3ar293Z2Q3qfSsv355AW5Muevjo='
img_str = getByte('mask.jpg')
CurTime = datetime.datetime.now()
requestsss={
    'datastreams':[
        {'id':"pic","datapoints":[{"at":CurTime.isoformat(),"value":img_str}]}
        ]
    }
req = json.dumps(requestsss)
#print(req)
data=bytes(req,'utf-8')
request = urllib.request.Request(url, data)
request.add_header('api-key', APIKEY)
request.get_method = lambda:'POST'          # 设置HTTP的访问方式
request = urllib.request.urlopen(request)
print(request.read())
